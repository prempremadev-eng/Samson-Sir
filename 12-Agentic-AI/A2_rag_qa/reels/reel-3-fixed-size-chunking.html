4def fixed_size_chunks(documents):
5    splitter = CharacterTextSplitter(

7        chunk_size=500,
8        chunk_overlap=50,

10    chunks = []
11    for doc in documents:
12        pieces = splitter.split_text(doc["text"])
13        for piece in pieces:
14            chunks.append({"source": doc["source"], "text": piece})
15    return chunks
